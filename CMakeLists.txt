# ==================================================================================================
# CoralUI
# ==================================================================================================
# The application UI framework: the MVVM layer and the UI toolkit. It depends on CoralCore and
# CoralGraphics and deliberately does not use CoralGraphics' optional effect module, CoralMedia,
# Playback or Compose.
#
# As a subproject the application has already added CoralCore and CoralGraphics. A standalone
# checkout initializes the external/ submodules for them and builds them from this root.

cmake_minimum_required(VERSION 3.19)
# C is required for the Wayland protocol sources generated by wayland-scanner.
project(CoralUI LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(cmake/CoralBuild.cmake)
coral_apply_global_build_settings()

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(CORALUI_TOP_LEVEL ON)
else ()
  set(CORALUI_TOP_LEVEL OFF)
endif ()

if (CORALUI_TOP_LEVEL)
  if (NOT DEFINED IS_HOST_PLATFORM)
    set(IS_HOST_PLATFORM TRUE)
  endif ()
  if (NOT DEFINED CORAL_BUILD_TESTS)
    set(CORAL_BUILD_TESTS ON CACHE BOOL "Build the CoralUI tests")
  endif ()
  set(CORALUI_BUILD_TESTS ON CACHE BOOL "Build the CoralUI tests")
else ()
  if (NOT DEFINED CORAL_BUILD_TESTS)
    set(CORAL_BUILD_TESTS OFF CACHE BOOL "Build the CoralUI tests")
  endif ()
  set(CORALUI_BUILD_TESTS ${CORAL_BUILD_TESTS} CACHE BOOL "Build the CoralUI tests")
endif ()
set(CORAL_BUILD_TESTS ${CORALUI_BUILD_TESTS})
if (CORAL_BUILD_TESTS)
  enable_testing()
endif ()

# The module package configs carry the framework version. The application root defines
# CORAL_VERSION; a standalone checkout reads this repository's VERSION file.
if (NOT DEFINED CORAL_VERSION)
  if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" CORAL_VERSION LIMIT_COUNT 1)
    string(STRIP "${CORAL_VERSION}" CORAL_VERSION)
  endif ()
  if (NOT DEFINED CORAL_VERSION OR CORAL_VERSION STREQUAL "")
    set(CORAL_VERSION "0.0.0")
  endif ()
endif ()

# The reference fonts ship with the examples. A standalone checkout uses them as the default test
# data root so the unit suites run from a fresh clone; the application passes its own assets/.
if (CORAL_BUILD_TESTS AND NOT DEFINED CORAL_TEST_DATA_DIR AND
    EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/fonts/NotoSansSC-Regular.otf")
  set(CORAL_TEST_DATA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/examples" CACHE PATH
      "Assets used by the CoralUI tests and benchmarks")
endif ()

# The reference fonts and test fixtures are an explicit input for a tree that keeps them elsewhere:
# the application passes its assets/ directory, and a standalone checkout can point at any directory
# that contains the fixtures it needs.
if (CORAL_BUILD_TESTS AND NOT DEFINED CORAL_TEST_DATA_DIR)
  message(STATUS
      "CoralUI: CORAL_TEST_DATA_DIR is not set; the font-dependent tests need "
      "-DCORAL_TEST_DATA_DIR=<assets>")
endif ()

# Dependencies. A standalone clone uses the external/ submodules; a checkout beside the other
# frameworks (the application's frameworks/ tree or the workgroup layout) reuses the sibling modules.
# The application root adds its own targets first, so this include is a no-op there.
include("${CMAKE_CURRENT_LIST_DIR}/cmake/CoralUIDependencies.cmake")

if (NOT TARGET graphics)
  message(FATAL_ERROR "CoralUI needs CoralGraphics; add it before CoralUI.")
endif ()

add_subdirectory(modules/mvvm)
add_subdirectory(modules/ui)

# ==================================================================================================
# Examples
# ==================================================================================================
# A standalone configure builds the demo and verification programs a client would write. An
# application that adds this framework as a subproject gets libraries and tests only, and keeps its
# own products; the examples project is also configurable on its own (`cmake -S examples`), which is
# what an IDE opens when working on a demo.
if (CORALUI_TOP_LEVEL)
  add_subdirectory(examples)
endif ()
